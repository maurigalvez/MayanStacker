# Play Core's listener base. InstallStateUpdatedListener extends this, and the flexible
# update's download callbacks are dispatched through the parent's onStateUpdate(Object).
# Our side of that interface is an AndroidJavaProxy, which binds by name at runtime, so if
# R8 renames the parent method the proxy is never called: no progress, no "downloaded",
# and the update silently stalls. Google's own common.txt marks this rule always-include.
-keep class com.google.android.play.core.listener.StateUpdatedListener {
    public abstract void onStateUpdate(java.lang.Object);
}
